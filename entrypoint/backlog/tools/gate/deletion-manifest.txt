# deletion-manifest.txt — the exact file list for the hard cutover.
#
# Hand-deleting ~73 files is how a survivor gets destroyed by accident. This
# manifest is the reviewable source of truth; `wipe.mjs --check` verifies every
# PRESERVE path still exists and every DELETE path is still present before any
# removal happens, and refuses to run if the two sets intersect.
#
# PRESERVE lines are listed FIRST and deliberately, because the whole point is
# that `src/store/` is NOT deleted wholesale: serve-lock.ts (290 lines) and
# signal-cleanup.ts (114) have zero internal imports, are needed by server.ts
# and the CLI, and are merely filed in the wrong directory. Deleting the
# directory destroys 404 lines of working infrastructure.

PRESERVE store/type-policy.ts
PRESERVE store/serve-lock.ts
PRESERVE store/serve-lock.spec.ts
PRESERVE store/signal-cleanup.ts
PRESERVE store/signal-cleanup.spec.ts

# --- G2-G5 corrections: the store factory the NEW layer opens through, and its
# --- transitive deps. Found by computing survivor->doomed import edges; the
# --- earlier containment check only looked for importers OUTSIDE the package.
PRESERVE store/graph-backlog-store.ts
PRESERVE store/semantic-search.ts
PRESERVE store/immediate-retry.ts
PRESERVE store/embed-queue.ts
PRESERVE store/mutate-metadata.ts

# --- specs of the newly-preserved modules. Preserving a module while deleting
# --- its spec silently drops coverage while the suite stays green.
# --- graph-backlog-store.spec + rag-optional-deps.spec need ZERO rework;
# --- semantic-search.spec needs one import repoint (model.js -> envelope.js).
PRESERVE store/graph-backlog-store.spec.ts
PRESERVE store/semantic-search.spec.ts
PRESERVE store/rag-optional-deps.spec.ts

# --- RESEED-REQUIRED: these two are the only behavioural coverage of the
# --- preserved embedding path, but they seed through the doomed store/crud.ts
# --- (createItemNode/updateItemNode/softDeleteItemNode). They are preserved and
# --- reseeded onto write/create-issue.ts IN the cutover commit, never deleted.
PRESERVE store/rag-e2e.spec.ts
PRESERVE store/embed-write-path.spec.ts

# --- G4: apigen mount wiring (cliPlugin), NOT hand-rolled v1 argv. Zero
# --- occurrences of humanId/markdown/ops-v1/idOverride/repoKey. SPEC 6.7
# --- mandates the CLI keep the "unchanged mechanism, new operation list".
# --- Deleting it strands cli.spec.ts, which this manifest already preserves.
PRESERVE cli.ts

# --- whole directory: the v2 namespace collapses into the flat layer ---
DELETE_GLOB v2/*.ts

# --- store/: everything except the five PRESERVE paths above ---
# store/mapping.ts stays DELETE: embed-queue needs only `stripContentMarker`
# (a one-line regex, rehomed into embed-queue.ts) and mutate-metadata needs
# only `BacklogNodeMeta` as a DEFAULT TYPE PARAM (retyped Record<string,unknown>).
# Its five v1 model types (BacklogItem/BacklogStatus/Citation/Note/Priority) die with it.
DELETE_GLOB store/*.ts

# --- top-level modules ---

DELETE client.ts
DELETE ops-v1.ts

DELETE migration-admin.ts
DELETE migration-phase.ts

# --- specs bound to the deleted modules ---
