/** Splits argv at the first literal `--`. ### `wrapper/attributes-source.ts`
```ts
import type { Attributes } from '@adhd/policy-core-client';
export interface IAttributesSource { read(): Promise<Attributes | undefined>; readonly origin: 'flag' | 'env' | 'none' }
/** D-06: flag value, else env value (ADHD_POLICY_ATTRIBUTES passed IN by the caller as `env`, this file reads no process.env), else none. A value starting with `@` is a file path read on EVERY read() (so a rotating identity file is honoured). JSON must be an object whose values are string|number|string[]; otherwise ApiError('invalid_argument', 'attributes: ...') thrown at read() — the verbs call read() once eagerly so a bad value refuses to start (exit 2, B3.6). Flag and env are NEVER merged. */
export function createAttributesSource(input: { flag?: string; env?: string; cwd: string; readFile?: (path: string) => Promise<string> }): IAttributesSource;
```

### `wrapper/verbs/run.ts`
```ts
export interface IVerbDeps { cwd: string; env: Readonly<Record<string, string | undefined>>; launcher?: IProcessLauncher; readFile?: (p: string) => Promise<string>; stdout: (s: string) => void; stderr: (s: string) => void; signal?: AbortSignal }
/** `run bin:<cmd> [--spec f]... [--adhd-attributes v] -- args...`. Steps: splitPolicyArgv; parseSourceSpec (kind must be bin else invalid_argument); createAttributesSource + one eager read; buildGate; resolveSource(spec,'run'); `createInvoker(usePlugins layers)` then invoke(opId, call{operation, ctx: <empty LayerContext>, envelope:{}, domainArgs:{argv: childArgs, cwd}, transport:'cli', signal}, {fns, schemas, createClient}); child stdout/stderr are forwarded to deps.stdout/stderr; returns the child's exit code, or 3 when the Layer threw permission_denied (message to stderr, child NOT started), 2 on invalid_argument, 1 otherwise (CLI_EXIT_CODE). dispose() in finally. */
export async function runVerb(argv: readonly string[], deps: IVerbDeps): Promise<number>;
```
