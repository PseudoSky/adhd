### `src/plugin.ts`
```ts
import type { UsePlugin } from '@adhd/apigen-engine-runtime';
import type { IPolicyAskHandler, IPolicyDecider } from './ports.js';
export interface IPolicyLayerOptions { readonly decider: IPolicyDecider | string; readonly deciderOptions?: Readonly<Record<string, unknown>>; readonly ask?: IPolicyAskHandler; readonly timeoutMs?: number }
/** Layer algorithm (LP-06..LP-17), one call: 1 resolve decider once (memoised promise; a load failure memoises a rejected state => every call denies); 2 map call.transport — 'http'|'grpc'|'mcp'|'cli' pass through; any other value => deny 'unsupported transport'; 3 request = {tool: call.operation.id, args: call.domainArgs, envelope: call.envelope, transport, attributes: validated call.envelope.attributes (only string|number|string[] values; invalid shape => undefined)}; 4 decision = await decide(request, call.signal) raced against timeoutMs (expiry => deny); 5 validate result shape (decision in allow|ask|deny; set is a plain object when present) else deny; 6 allow => merge `set` over call.domainArgs (inside the guarded section; a non-extensible target => ApiError('internal') => standard deny) then next(); 7 ask => ask handler 'allow-once' => next(), anything else/absent/throws => deny; 8 deny => throw new ApiError('permission_denied', reason ?? 'denied by policy', {code:'policy-denied'}). Holds no state between calls (LP-13). Streaming results are returned untouched (LP-12). */
export function createPolicyLayerPlugin(options: IPolicyLayerOptions): UsePlugin;
/** Default export loaded by `--use @adhd/apigen-plugin-policy`: id 'policy', capabilities.layer delegates to a plugin built from the captured options (throws ApiError('internal', 'apigen-plugin-policy: no decider configured') => treated as deny if options never arrived), capabilities.mount.operations(descriptor, opts) stores `opts` via state.set and returns []. */
declare const plugin: UsePlugin;
export default plugin;
```

## Review fixes (Phase B, L1..L7 of `REVIEW-FIXES.md`)
- **L1** `readAttributes` rejects an attributes object with an own `__proto__` key (no attributes reach the decider) and builds the frozen result on a null prototype via `defineProperty`.
- **L2** `onError(error, { phase })`, phases `load | decide | timeout | shape | merge`; default writes one stderr line (phase and error class only). The client still sees the generic `policy decision failed`.
- **L3 (security) finding: the envelope channel is kept.** A client cannot put `attributes` in `call.envelope`: `computeEnvelopeFields` (`packages/apigen/apigen-engine-runtime/src/lib/op-plan.ts:240-266`) derives `plan.envelope` from the composed input schema's top-level properties other than `data`, which `composeSchemas` fills only from plugins' declared `envelope` fragments (`apigen-core-client/src/lib/compose-schemas.ts:196-204`, with `additionalProperties:false` at `:257`). Every host copies a value only for a field in `plan.envelope`: mcp `apigen-plugin-mcp/src/lib/run.ts:287-291` (`_meta`), fastify `apigen-plugin-api-fastify/src/lib/run.ts:232-236` (headers), express `apigen-plugin-api-express/src/lib/run.ts:190-194`, cli `apigen-plugin-cli-output/src/lib/run.ts:367-446` (flags and env vars from the same table). `ctx` is `new LayerContext()` per call (`dispatch-for-plan.ts:93`), never client-fed. The policy plugin declares no envelope field. Residual: a second `--use` plugin declaring an envelope field named `attributes` would open the channel (documented in the README and D-18); the wrapper's attributes layer additionally overwrites or deletes the key on every call (`entrypoint/policy/src/wrapper/attributes-layer.ts`). Therefore no change to the wrapper, T09 or the key name.
- **L4** The default export's `mount.operations` ignores calls with undefined/null options once a config is captured; documented as one instance per process.
- **L5** `timeoutMs` must be a finite number > 0 or construction/capture throws `ApiError('invalid_argument')`; expiry aborts the `AbortSignal` passed to `decide` (combined with `call.signal`).
- **L6** The decider receives frozen shallow copies of `domainArgs` and `envelope`; the `set` merge runs inside the guarded section and throws `ApiError('internal')` for a non-extensible `domainArgs`, which denies with the standard shape.
- **L7** `vite.config.mts` test environment `node`; `package.json` gains `module`, `exports`, `files` and source-relative `./dist/...` entry paths (the manifest executor rebases them for the published dist); `tsconfig.json` gains `strict` and `ignoreDeprecations` like the sibling plugins.
