### `sources/types.ts`
```ts
import type { Operation, composeSchemas } from '@adhd/apigen-core-client';
export type SourceSpec =
  | { readonly kind: 'file'; readonly path: string }
  | { readonly kind: 'remote'; readonly url: string }
  | { readonly kind: 'bin'; readonly command: string };
export type SourceIntent = 'run' | 'serve';
export interface IResolvedSource {
  readonly sourceId: string;                         // 'bin:touch' | 'bin:mcp-server-filesystem'
  readonly operations: Operation[];
  readonly pkg: { id: string; version: string; schemas: ReturnType<typeof composeSchemas>; importPath: string; fns: Record<string, (...a: unknown[]) => unknown>; createClient: () => Promise<unknown> };
  dispose(): Promise<void>;                          // kills any started process; idempotent
}
export class SourceError extends Error { constructor(readonly code: 'unsupported' | 'handshake-failed' | 'spawn-failed' | 'invalid-spec', message: string) }
```
### `sources/parse-source.ts`
```ts
/** 'bin:<command>' -> bin; 'remote:<url>' -> remote (http/https only, else invalid-spec); anything else -> file. The command must be non-empty and contain no whitespace-separated args (SX-02 analogue: a name or path, never a command line) else SourceError('invalid-spec'). */
export function parseSourceSpec(text: string): SourceSpec;
```
### `sources/process.ts`
```ts
export interface IProcessSpec { readonly command: string; readonly args: readonly string[]; readonly cwd: string; readonly env?: Readonly<Record<string, string>> }
export interface IProcessHandle {
  readonly stdin: { write(chunk: string): void; end(): void };
  onStdout(cb: (chunk: string) => void): void; onStderr(cb: (chunk: string) => void): void;
  exited: Promise<{ code: number | null; signal: string | null }>;
  kill(signal?: 'SIGTERM' | 'SIGKILL'): void;
}
export interface IProcessLauncher { start(spec: IProcessSpec): Promise<IProcessHandle> }   // rejects SourceError('spawn-failed')
/** Default: child_process.spawn with shell:false, stdio pipes, env = process.env overlaid by spec.env. */
export const nodeProcessLauncher: IProcessLauncher;
```
### `sources/argv-operation.ts`
```ts
import type { Operation } from '@adhd/apigen-core-client';
/** D-07. id = basename of command without extension (e.g. 'touch'). input schema: {type:'object', required:['argv','cwd'], additionalProperties:false, properties:{argv:{type:'array',items:{type:'string'}}, cwd:{type:'string'}}}; output schema {exitCode:number, stdout:string, stderr:string}; kind 'action', async true, streaming false, safe false, hasCtx false. */
export function buildArgvOperation(command: string): { operation: Operation; fnName: string };
/** fns entry: starts the process via `launcher` with {command, args: input.argv, cwd: input.cwd}, collects stdout/stderr (UTF-8 strings, 1 MiB cap each, truncation flagged by a trailing marker), resolves with {exitCode: code ?? -1, stdout, stderr}. Never invoked during extraction (SRC-05/PS-28). */
export function createArgvInvoker(command: string, launcher: IProcessLauncher): (input: { argv: string[]; cwd: string }) => Promise<{ exitCode: number; stdout: string; stderr: string }>;
```

### `sources/mcp-source.ts`
```ts
/** D-08. Starts the process, connects an SDK `Client` over createLauncherTransport, performs `initialize` bounded by `handshakeTimeoutMs` (default 10000), lists tools (follow cursors), builds one Operation per tool: id `<server>.<tool>` with server = sanitized `serverInfo.name` (lowercase, [a-z0-9_-], fallback basename of command); input = tool.inputSchema (object schema forced: non-object => SourceError('unsupported')); output = {} ; safe = annotations.readOnlyHint === true; hints ride in a side map returned for `draft`: `hints[opId] = {readOnly, destructive}`. fns[opId](input) = client.callTool({name, arguments: input}) mapped to its result content; isError results are returned as data (the upstream's own error), never thrown. dispose() closes the client and kills the process. Handshake failure/timeout => SourceError('handshake-failed') and the process is killed (never falls back to argv). */
export async function resolveMcpStdioSource(spec: { command: string; args: readonly string[]; cwd: string; env?: Record<string,string> }, deps: { launcher: IProcessLauncher; handshakeTimeoutMs?: number }): Promise<IResolvedSource & { hints: Readonly<Record<string, { readOnly?: boolean; destructive?: boolean }>> }>;
```
### `sources/resolve-source.ts`
```ts
export interface IResolveDeps { readonly launcher?: IProcessLauncher; readonly cwd: string; readonly args?: readonly string[]; readonly env?: Record<string,string>; readonly handshakeTimeoutMs?: number }
/** bin+run -> argv operation (process NOT started until the operation is invoked); bin+serve -> resolveMcpStdioSource (starts the process now: explicit serve); remote -> SourceError('unsupported', 'remote sources are not implemented'); file -> SourceError('unsupported', 'file sources are resolved by apigen-cli'). */
export async function resolveSource(spec: SourceSpec, intent: SourceIntent, deps: IResolveDeps): Promise<IResolvedSource>;
```

### T08b — `sources/mcp-passthrough-server.ts` (added after discovery finding B-1; D-21)
`mcpPlugin.run` renames tools to `<namespace>_<path>`, wraps input as `{data:{...}}` and rewraps results, so an unmodified MCP host cannot use it to front an upstream server. The gateway therefore ships its own thin server on the SDK, dispatching through the same invoker and Layer stack:
```ts
import type { UsePlugin } from '@adhd/apigen-engine-runtime';
import type { IResolvedSource } from './types.js';
export interface IPassthroughServerOptions {
  readonly source: IResolvedSource & { hints?: Readonly<Record<string, { readOnly?: boolean; destructive?: boolean }>> };
  readonly usePlugins: readonly UsePlugin[];     // layer capabilities only are used; composed with createInvoker, first = outermost
  readonly serverInfo: { name: string; version: string };
  readonly transport: import('@modelcontextprotocol/sdk/shared/transport.js').Transport;   // stdio in production, in-memory in tests
  readonly signal: AbortSignal;
}
/** SDK `Server` advertising exactly the upstream tools: names are the upstream names verbatim, inputSchema verbatim, annotations from hints. `tools/call` -> createInvoker(layers)(fnName, Call{operation, ctx: new LayerContext(), envelope:{}, domainArgs: request.arguments ?? {}, transport:'mcp', signal}, {fns, schemas, createClient}). The Layer sees `operation.id` (`<server>.<tool>`) and the FLAT upstream arguments (so permissions are written against real argument names). Layer refusal (ApiError permission_denied) => result {isError:true, content:[{type:'text', text: <reason>}]}; the upstream is NOT called. Success => the upstream CallToolResult returned verbatim (content, structuredContent, isError). Unknown tool => JSON-RPC error. Runs until `signal` aborts, then closes the server and calls source.dispose(). */
export async function serveMcpPassthrough(options: IPassthroughServerOptions): Promise<void>;
```
Export from `src/index.ts`. T09's `serveVerb` uses this for `bin:` MCP sources (not `mcpPlugin.run`). The operation `id` stays `<server>.<tool>` (D-08) and permission `tool:` keys use it.
