### `spec/results.ts` (operation result types; the spec left these open)
```ts
import type { Attributes, AttributeKind, Effect, FilterExpression, IPolicyDocument, LayerName } from './policy.js';
import type { ICompiledPolicy } from './compiled.js';
import type { IPolicyIssue } from './decision.js';

/** Result of `compile`. `policy` is present exactly when `ok` is true. */
export interface ICompileSummary {
  readonly ok: boolean;
  readonly compilerVersion: string;
  readonly inputHash?: string;
  readonly contentHash?: string;
  readonly permissionCount: number;
  readonly defaultsCount: number;
  readonly attributes: Readonly<Record<string, AttributeKind>>;
  readonly errors: readonly IPolicyIssue[];
  readonly warnings: readonly IPolicyIssue[];
  readonly policy?: ICompiledPolicy;
}

/** Result of `verify`: every defect, never just the first (D-05). */
export interface IVerifyReport {
  readonly ok: boolean;                 // true iff errors is empty
  readonly errors: readonly IPolicyIssue[];
  readonly warnings: readonly IPolicyIssue[];
  readonly sources: readonly string[];  // layer sources that were read, in layer order
}

/** Result of `show`. */
export interface IEffectivePolicy {
  readonly sources: readonly { readonly layer: LayerName; readonly source: string }[];
  readonly attributes: Attributes;                         // effective, after defaults
  readonly provenance: Readonly<Record<string, string>>;
  readonly permissions: readonly {
    readonly id: string; readonly layer: LayerName; readonly tools: readonly string[]; readonly effect: Effect;
  }[];
}

/** Result of `doctor`. `settings` shows each resolved setting and the cascade level it came from. */
export interface IDoctorReport {
  readonly ok: boolean;
  readonly checks: readonly { readonly name: string; readonly ok: boolean; readonly detail: string }[];
  readonly settings: Readonly<Record<string, { readonly value: unknown; readonly source: string }>>;
  readonly policySource: string;        // path of the discovered layer file(s) or 'built-in'
}

/** One redacted audit record (AU-01). Attribute VALUES are never recorded, names only. */
export interface IAuditEntry {
  readonly at: number;                  // ctx.now()
  readonly tool: string;
  readonly effect: Effect;
  readonly matched: readonly string[];
  readonly denialCode?: 'explicit-deny' | 'no-match' | 'eval-error' | 'runtime-unavailable';
  readonly attributeNames: readonly string[];
  readonly transport?: string;
}

/** Plain-data view of one foreign operation; the client imports no apigen type (SRC-11). */
export interface IDraftOperation {
  readonly name: string;
  readonly args: readonly {
    readonly name: string; readonly type: string; readonly kind?: string; readonly provenance?: 'declared' | 'inferred';
  }[];
  readonly hints?: { readonly readOnly?: boolean; readonly destructive?: boolean };
}
export interface IDraftOptions {
  readonly source: string;                         // label used in comments/messages
  readonly secretPatterns?: readonly string[];     // replaces the built-in secret list when set
  readonly readRootsAttribute?: string;            // default 'read_roots'
  readonly writeRootsAttribute?: string;           // default 'write_roots'
}
export interface IInferredEntry {
  readonly operation: string; readonly argument: string; readonly kind: string; readonly reason: string;
}
export interface IDraftPolicy {
  readonly document: IPolicyDocument;
  readonly yaml: string;
  readonly inferred: readonly IInferredEntry[];
}
```

### `spec/schema/policy-schema.ts` and `spec/schema/compiled-schema.ts`
Each exports one `const` JSON Schema (draft 2020-12, `as const`): `POLICY_SCHEMA`, `COMPILED_SCHEMA`. POLICY_SCHEMA rules: `version` required and `const: 1`; `additionalProperties: false` at every object level (no silent ignore); `permissions[].tool` string or non-empty array of non-empty strings; `effect` enum `allow|ask|deny`; `attributes` values enum `name|names|paths|number`; `runtime` values enum `number|name`; `defaults[].set` required. COMPILED_SCHEMA mirrors `ICompiledPolicy` with `schemaVersion: const 1`, `contentHash`/`inputHash` pattern `^[0-9a-f]{64}$`. `$id`s: `https://adhd.dev/schema/policy-1.json`, `.../compiled-1.json`.
